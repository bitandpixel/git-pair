func repo(f *gittest.Fixture) *git.Repo { return &git.Repo{Dir: f.Dir()} }

func archive(f *gittest.Fixture, id, sha string) {
	f.MustGit("update-ref", "refs/git-pair/changesets/"+id+"/archive", sha)
}

func integrate(f *gittest.Fixture, id, sha string) {
	f.MustGit("update-ref", "refs/git-pair/changesets/"+id+"/integration", sha)
}

func active(t *testing.T, f *gittest.Fixture) discovery.Result {
	t.Helper()
	res, err := discovery.Active(context.Background(), repo(f), "HEAD")
	if err != nil {
		t.Fatalf("Active: %v", err)
	}
	return res
}

// The rule working as intended: a stacked changeset's init commit sits between its parent's
// archive tip and HEAD, so the nearer one is chosen.
func TestStackedPicksTheNearestArchive(t *testing.T) {
	f := gittest.New(t)
	f.Commit("seed", gittest.WithFile("a.txt", "a\n"))

	f.CreateBranch("booking")
	parent := f.Commit("implement booking", gittest.WithFile("booking.txt", "1\n"))
	archive(f, "booking", parent)

	f.CreateBranch("booking-tests")
	child := f.Commit("add booking tests", gittest.WithFile("booking_test.txt", "1\n"))
	archive(f, "booking-tests", child)
	f.Commit("more tests", gittest.WithFile("b.txt", "b\n"))

	got := active(t, f)
	if got.Candidate == nil || got.Candidate.ID != "booking-tests" {
		t.Fatalf("active = %+v, want booking-tests (nearest archive tip)", got.Candidate)
	}

	f.SwitchTo("booking")
	got = active(t, f)
	if got.Candidate == nil || got.Candidate.ID != "booking" {
		t.Errorf("on the parent branch active = %+v, want booking", got.Candidate)
	}
}

// "A branch created from an existing changeset naturally continues that changeset" and one
// movable archive ref cannot serve two branches that both keep working. The child's
// `change ready` moves the shared ref off the parent's line of history, and the parent
// then has no ancestor archive ref at all.
func TestDivergedBranchesLoseTheirChangeset(t *testing.T) {
	f := gittest.New(t)
	f.Commit("seed", gittest.WithFile("a.txt", "a\n"))

	f.CreateBranch("booking")
	init := f.CommitChangeset("booking", "main")
	archive(f, "booking", init)

	// The child continues the same changeset without initializing a new one.
	f.CreateBranch("booking-experiment")
	f.Commit("experiment on the same changeset", gittest.WithFile("experiment.txt", "x\n"))
	// `change ready` on the child moves the one archive ref to the child's commit.
	child := f.Head()
	archive(f, "booking", child)

	f.SwitchTo("booking")
	got := active(t, f)
	if got.Candidate != nil {
		t.Errorf("parent branch resolved %s; expected it to keep its changeset", got.Candidate.ID)
	}
	t.Logf("parent branch after the child moved the shared ref: uninitialized (no ancestor archive ref)")

	// And the directory is still sitting on the parent branch.
	if _, err := os.Stat(filepath.Join(f.Dir(), "changesets", "booking")); err != nil {
		t.Errorf("expected the changeset directory to still be on the parent branch: %v", err)
	}
}

// Two changesets integrated with a merge commit are both ancestors of the branch that
// follows, and neither is an ancestor of the other. Unless integration has been recorded,
// discovery has no way to tell "both landed" from "two active changesets in a strange
// shape", and rule 4 makes that a failure for every branch built afterwards.
func TestMergeCommitIntegrationWithoutRecordIsAmbiguous(t *testing.T) {
	f := gittest.New(t)
	f.Commit("seed", gittest.WithFile("a.txt", "a\n"))

	f.CreateBranch("one")
	one := f.Commit("work one", gittest.WithFile("one.txt", "1\n"))
	archive(f, "one", one)

	f.SwitchTo("main")
	f.CreateBranch("two")
	two := f.Commit("work two", gittest.WithFile("two.txt", "2\n"))
	archive(f, "two", two)

	f.SwitchTo("main")
	f.MustGit("merge", "--no-ff", "-m", "merge one and two", "one", "two")

	f.CreateBranch("three")
	f.Commit("later work", gittest.WithFile("three.txt", "3\n"))

	got := active(t, f)
	if got.Candidate == nil && len(got.Ambiguous) == 0 {
		t.Fatal("expected ambiguity or a candidate, got neither")
	}
	if len(got.Ambiguous) > 0 {
		ids := make([]string, len(got.Ambiguous))
		for i, c := range got.Ambiguous {
			ids[i] = fmt.Sprintf("%s@%d", c.ID, c.Dist)
		}
		t.Logf("a branch off main after two merge-commit landings is AMBIGUOUS: %v", ids)
	} else {
		t.Logf("resolved %s at distance %d", got.Candidate.ID, got.Candidate.Dist)
	}

	// Recording integration for both makes the same question answerable.
	integrate(f, "one", f.RevParse("main"))
	integrate(f, "two", f.RevParse("main"))
	if got = active(t, f); got.Candidate != nil {
		t.Errorf("after integration was recorded, active = %s; want uninitialized", got.Candidate.ID)
	} else if len(got.Ambiguous) > 0 {
		t.Errorf("after integration was recorded, still ambiguous: %v", got.Ambiguous)
	} else {
		t.Logf("with both integration refs recorded, the same branch is cleanly uninitialized")
	}
}

// The same two landings, done with squash instead of a merge commit: the archive targets
// are no longer ancestors, so nothing is ambiguous. Discovery's behaviour therefore depends
// on the integration strategy, which §15 asks git-pair to be independent of.
func TestSquashIntegrationIsInvisibleToDiscovery(t *testing.T) {
	f := gittest.New(t)
	f.Commit("seed", gittest.WithFile("a.txt", "a\n"))

	f.CreateBranch("one")
	one := f.Commit("work one", gittest.WithFile("one.txt", "1\n"))
	archive(f, "one", one)

	// A squash landing: one new commit on main with the same content, no ancestry.
	f.SwitchTo("main")
	f.Write("one.txt", "1\n")
	f.Commit("squash: work one")

	f.CreateBranch("three")
	f.Commit("later work", gittest.WithFile("three.txt", "3\n"))

	got := active(t, f)
	if got.Candidate != nil || len(got.Ambiguous) > 0 {
		t.Errorf("a squash-merged changeset still resolves as active: %+v %v", got.Candidate, got.Ambiguous)
	}
	t.Logf("squash landings leave no ancestor archive ref: discovery is strategy-dependent")
}

// Nothing deletes refs, so an initialization that was never readied or abandoned stays a
// candidate forever — and it is an ancestor of every branch built from where it happened.
func TestAnUnabandonedInitOwnsEveryDescendant(t *testing.T) {
	f := gittest.New(t)
	f.Commit("seed", gittest.WithFile("a.txt", "a\n"))

	// A stray `change init` on the trunk: never readied, never abandoned.
	stray := f.Commit("git-pair: initialize changeset scratch", gittest.WithFile("scratch-note.md", "# scratch\n"))
	archive(f, "scratch", stray)

	f.CreateBranch("real-work")
	f.Commit("actual work", gittest.WithFile("b.txt", "b\n"))

	got := active(t, f)
	if got.Candidate == nil || got.Candidate.ID != "scratch" {
		t.Fatalf("active = %+v, want the demonstration that the stray init owns this branch", got.Candidate)
	}
	t.Logf("a forgotten `change init` on the trunk is the active changeset for every later branch (distance %d)", got.Candidate.Dist)

	// Abandoning it releases the branches — but only once someone notices to do it.
	f.MustGit("update-ref", "-d", "refs/git-pair/changesets/scratch/archive")
	if got = active(t, f); got.Candidate != nil {
		t.Errorf("after deleting the ref, active = %s; want uninitialized", got.Candidate.ID)
	}
}

// The cost of discovery is per archive ref, not per branch or per changeset directory:
// one ref listing, then an ancestry test, a terminal check and a distance count each.
func TestCostPerArchiveRef(t *testing.T) {
	f := gittest.New(t)
	f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
	base := f.Head()
	for i := 0; i < 300; i++ {
		f.Commit(fmt.Sprintf("work %d", i), gittest.WithFile(fmt.Sprintf("f%d.txt", i), "x\n"))
		archive(f, fmt.Sprintf("cs-%03d", i), f.Head())
	}
	_ = base
	out, err := exec.Command("git", "-C", f.Dir(), "rev-list", "--count", "HEAD").Output()
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("HEAD history: %s commits", trim(string(out)))

	spawns := measureSpawns(t, f, func() {
		if _, err := discovery.Active(context.Background(), repo(f), "HEAD"); err != nil {
			t.Fatal(err)
		}
	})
	t.Logf("300 archive refs, 301-commit history: %d git invocations per discovery", spawns)
}

func trim(s string) string {
	for len(s) > 0 && (s[len(s)-1] == '\n' || s[len(s)-1] == ' ') {
		s = s[:len(s)-1]
	}
	return s
}

func measureSpawns(t *testing.T, f *gittest.Fixture, fn func()) int {
	t.Helper()
	dir := t.TempDir()
	log := filepath.Join(dir, "spawns.log")
	shim := filepath.Join(dir, "git")
	body := fmt.Sprintf("#!/bin/sh\nprintf 'x\\n' >> %s\nexec %s \"$@\"\n", shellQuote(log), gitBinary())
	if err := os.WriteFile(shim, []byte(body), 0o755); err != nil {
		t.Fatal(err)
	}
	t.Setenv("PATH", dir+string(os.PathListSeparator)+os.Getenv("PATH"))
	fn()
	data, err := os.ReadFile(log)
	if err != nil {
		return -1
	}
	return len(data)
}

func gitBinary() string {
	out, err := exec.Command("which", "git").Output()
	if err != nil {
		return "git"
	}
	return trim(string(out))
}

func shellQuote(s string) string { return "'" + s + "'" }

// The two fixtures that refuse to answer under the rules as written, plus the diverged
// parent, under the amended rule.
func TestLandedRuleFixesTheAmbiguousFixtures(t *testing.T) {
	t.Run("merge landing without a record", func(t *testing.T) {
		f := gittest.New(t)
		f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
		f.CreateBranch("one")
		archive(f, "one", f.Commit("work one", gittest.WithFile("one.txt", "1\n")))
		f.SwitchTo("main")
		f.CreateBranch("two")
		archive(f, "two", f.Commit("work two", gittest.WithFile("two.txt", "2\n")))
		f.SwitchTo("main")
		f.MustGit("merge", "--no-ff", "-m", "merge one and two", "one", "two")
		f.CreateBranch("three")
		f.Commit("later work", gittest.WithFile("three.txt", "3\n"))

		got := landed(t, f)
		if got.Candidate != nil || len(got.Ambiguous) > 0 {
			t.Fatalf("got %+v, want uninitialized: both changesets have landed", got)
		}
		t.Logf("a branch off main after two unrecorded merge-commit landings is uninitialized")
	})

	t.Run("forgotten init on trunk", func(t *testing.T) {
		f := gittest.New(t)
		f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
		archive(f, "scratch", f.Commit("git-pair: initialize changeset scratch", gittest.WithFile("scratch-note.md", "# scratch\n")))
		f.CreateBranch("real-work")
		f.Commit("actual work", gittest.WithFile("b.txt", "b\n"))

		got := landed(t, f)
		if got.Candidate != nil {
			t.Fatalf("active = %s, want uninitialized", got.Candidate.ID)
		}
		t.Logf("the stray init no longer owns the branch: its archive target is on trunk")
	})

	t.Run("diverged parent keeps its changeset", func(t *testing.T) {
		f := gittest.New(t)
		f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
		f.CreateBranch("booking")
		archive(f, "booking", f.CommitChangeset("booking", "main"))
		f.CreateBranch("booking-experiment")
		f.Commit("experiment", gittest.WithFile("experiment.txt", "x\n"))
		archive(f, "booking", f.Head()) // the child's `change ready` moves the shared ref
		f.SwitchTo("booking")
		f.Commit("parent continues", gittest.WithFile("parent.txt", "p\n"))

		got := landed(t, f)
		if got.Candidate == nil || got.Candidate.ID != "booking" {
			t.Fatalf("active = %+v, want booking (HEAD is inside the archive's own future)", got.Candidate)
		}
		t.Logf("the parent branch keeps its changeset under the comparable-not-ancestor rule")
	})

	// The amendment does not rescue a squash landing: the archive targets are not on
	// trunk, so they look active forever unless integration is recorded.
	t.Run("squash landing still looks active", func(t *testing.T) {
		f := gittest.New(t)
		f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
		f.CreateBranch("one")
		archive(f, "one", f.Commit("work one", gittest.WithFile("one.txt", "1\n")))
		f.SwitchTo("main")
		f.CreateBranch("two")
		archive(f, "two", f.Commit("work two", gittest.WithFile("two.txt", "2\n")))
		f.SwitchTo("main")
		f.MustGit("merge", "--squash", "one")
		f.MustGit("commit", "-m", "squash one")
		f.CreateBranch("three")
		f.Commit("later work", gittest.WithFile("three.txt", "3\n"))

		got := landed(t, f)
		if got.Candidate == nil {
			t.Fatal("want the squash-landed changeset to still be a candidate")
		}
		t.Logf("after a squash landing the ref is not on trunk, so %s still looks active (distance %d)",
			got.Candidate.ID, got.Candidate.Dist)
	})
}

// The cost of the tree rule does not scale with the number of archive refs, which is the
// difference between a command that reads a repository and one that walks it.
func TestCostOfTreeRule(t *testing.T) {
	f := gittest.New(t)
	f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
	base := f.Head()
	f.CreateBranch("work")
	f.CommitChangeset("work", "main")
	for i := 0; i < 300; i++ {
		id := fmt.Sprintf("cs-%03d", i)
		archive(f, id, base)
	}
	archived := measureSpawns(t, f, func() { active(t, f) })
	tree := measureSpawns(t, f, func() { treeRule(t, f) })
	t.Logf("with 300 archive refs: rules as written = %d git invocations, tree rule = %d", archived, tree)
	if tree > 8 {
		t.Errorf("tree rule cost %d invocations, expected a handful", tree)
	}
}

// The stack order without any archive refs: the child's base names the parent's archive
// ref, so the parent is identifiable as a parent from metadata alone.
func TestTreeRuleOrdersAStackWithoutRefs(t *testing.T) {
	f := gittest.New(t)
	f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
	f.CreateBranch("booking")
	f.CommitChangeset("booking", "main")
	f.Commit("booking work", gittest.WithFile("booking.txt", "1\n"))
	f.CreateBranch("booking-tests")
	f.CommitChangeset("booking-tests", "refs/git-pair/changesets/booking/archive")
	f.Commit("test work", gittest.WithFile("booking_test.txt", "1\n"))

	// No archive() call in this fixture: the namespace is empty by construction.
	got := treeRule(t, f)
	if len(got.Ambiguous) == 0 {
		t.Fatalf("expected the ref-less stack to be ambiguous under distance alone, got %+v", got.Candidate)
	}
	ids := make([]string, len(got.Ambiguous))
	for i, c := range got.Ambiguous {
		ids[i] = c.ID
	}
	t.Logf("distance alone cannot order %v with no archive refs", ids)

	baseOf := discovery.BaseNames(context.Background(), repo(f), f.Head(), ids)
	var tips []string
	for _, id := range ids {
		if !discovery.IsBaseOf(id, baseOf, ids) {
			tips = append(tips, id)
		}
	}
	if len(tips) != 1 || tips[0] != "booking-tests" {
		t.Fatalf("tips = %v, want [booking-tests]", tips)
	}
	t.Logf("with base: read from the tree, the tip is %s and no refs were needed", tips[0])
}

func ids(hits []discovery.Hit) []string {
	out := make([]string, len(hits))
	for i, h := range hits {
		out[i] = fmt.Sprintf("%s/%s@%d", h.ID, h.Source, h.Dist)
	}
	return out
}

// The cases the ref fallback exists for, and the ones it must not reach into.
func TestCombinedRule(t *testing.T) {
	t.Run("directory deleted but the archive ref remains", func(t *testing.T) {
		f := gittest.New(t)
		f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
		f.CreateBranch("booking")
		init := f.CommitChangeset("booking", "main")
		archive(f, "booking", init)
		f.Commit("work", gittest.WithFile("booking.txt", "1\n"))
		f.MustGit("rm", "-r", "-q", "changesets/booking")
		f.Commit("drop the directory by accident", gittest.WithNoStage())

		got := combined(t, f)
		if got.Hit == nil || got.Hit.ID != "booking" || got.Hit.Source != discovery.FromRef {
			t.Fatalf("got %+v, want booking from the archive ref", got.Hit)
		}
		t.Logf("resolved %s from %s: %s", got.Hit.ID, got.Hit.Source, got.Hit.Note)
	})

	t.Run("a landed changeset is not resurrected by its own ref", func(t *testing.T) {
		f := gittest.New(t)
		f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
		f.CreateBranch("booking")
		init := f.CommitChangeset("booking", "main")
		archive(f, "booking", init)
		f.Commit("work", gittest.WithFile("booking.txt", "1\n"))
		f.SwitchTo("main")
		f.MustGit("merge", "--no-ff", "-m", "land booking", "booking")
		f.SwitchTo("booking")
		f.Commit("kept working after landing", gittest.WithFile("more.txt", "1\n"))

		got := combined(t, f)
		if got.Hit != nil || len(got.Ambiguous) > 0 {
			t.Fatalf("got %+v %v, want uninitialized: the changeset landed", got.Hit, ids(got.Ambiguous))
		}
		t.Logf("continued work on a landed changeset is uninitialized, not the old changeset")
	})

	t.Run("two unrelated changesets on one branch", func(t *testing.T) {
		f := gittest.New(t)
		f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
		f.CreateBranch("both")
		f.CommitChangeset("one", "main")
		f.CommitChangeset("two", "main")
		f.Commit("work", gittest.WithFile("x.txt", "1\n"))

		got := combined(t, f)
		if len(got.Ambiguous) != 2 {
			t.Fatalf("got %+v %v, want two ambiguous candidates", got.Hit, ids(got.Ambiguous))
		}
		t.Logf("unrelated siblings on one branch stay ambiguous: %v", ids(got.Ambiguous))
	})

	t.Run("init on trunk answers uninitialized", func(t *testing.T) {
		f := gittest.New(t)
		f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
		f.CommitChangeset("scratch", "main")
		f.CreateBranch("off-trunk")
		f.Commit("work", gittest.WithFile("b.txt", "b\n"))
		if got := combined(t, f); got.Hit != nil || len(got.Ambiguous) > 0 {
			t.Fatalf("got %+v, want uninitialized", got.Hit)
		}
	})

	t.Run("stacked child whose parent has landed", func(t *testing.T) {
		f := gittest.New(t)
		f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
		f.CreateBranch("booking")
		archive(f, "booking", f.CommitChangeset("booking", "main"))
		f.Commit("booking work", gittest.WithFile("booking.txt", "1\n"))
		f.CreateBranch("booking-tests")
		archive(f, "booking-tests", f.CommitChangeset("booking-tests", "refs/git-pair/changesets/booking/archive"))
		f.Commit("test work", gittest.WithFile("booking_test.txt", "1\n"))
		f.SwitchTo("main")
		f.MustGit("merge", "--no-ff", "-m", "land booking", "booking")

		f.SwitchTo("booking-tests")
		got := combined(t, f) // the child branch, after the parent landed
		if got.Hit == nil || got.Hit.ID != "booking-tests" {
			t.Fatalf("got %+v %v, want booking-tests", got.Hit, ids(got.Ambiguous))
		}
		t.Logf("the landed parent drops out and the child still resolves")
	})

	t.Run("the diverged pair still resolve", func(t *testing.T) {
		f := gittest.New(t)
		f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
		f.CreateBranch("booking")
		archive(f, "booking", f.CommitChangeset("booking", "main"))
		f.CreateBranch("booking-experiment")
		f.Commit("experiment", gittest.WithFile("experiment.txt", "x\n"))
		archive(f, "booking", f.Head())
		f.SwitchTo("booking")
		f.Commit("parent continues", gittest.WithFile("parent.txt", "p\n"))
		if got := combined(t, f); got.Hit == nil || got.Hit.ID != "booking" || got.Hit.Source != discovery.FromTree {
			t.Fatalf("got %+v, want booking from the tree", got.Hit)
		}
	})
}

// The fallback must not cost a walk over every changeset that has ever existed. On trunk
// the rank-1 set is empty by construction, so the whole rule costs a listing.
func TestCostOfCombinedRule(t *testing.T) {
	f := gittest.New(t)
	f.Commit("seed", gittest.WithFile("a.txt", "a\n"))
	base := f.Head()
	for i := 0; i < 300; i++ {
		archive(f, fmt.Sprintf("cs-%03d", i), base)
	}
	f.CreateBranch("work")
	f.CommitChangeset("work", "main")
	f.Commit("work", gittest.WithFile("w.txt", "1\n"))

	onBranch := measureSpawns(t, f, func() {
		if got := combined(t, f); got.Hit == nil {
			t.Fatal("expected the branch to resolve")
		}
	})
	f.SwitchTo("main")
	onTrunk := measureSpawns(t, f, func() {
		if got := combined(t, f); got.Hit != nil || len(got.Ambiguous) > 0 {
			t.Fatalf("trunk should be uninitialized, got %+v", got.Hit)
		}
	})
	t.Logf("300 archive refs: on a changeset branch = %d invocations, on trunk = %d", onBranch, onTrunk)
	if onTrunk > 12 {
		t.Errorf("the fallback is still iterating refs: %d invocations on trunk", onTrunk)
	}
}
